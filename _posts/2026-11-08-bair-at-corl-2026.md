<script>
(function () {
  "use strict";
  var root = document.querySelector(".corl");
  if (!root) return;
  var tools = root.querySelector(".corl-tools");
  var search = document.getElementById("corl-search"), type = document.getElementById("corl-type");
  var status = document.getElementById("corl-status"), empty = document.getElementById("corl-empty");
  var sessions = Array.prototype.slice.call(root.querySelectorAll(".corl-session"));
  tools.hidden = false;
  function norm(s) { return s.normalize("NFKD").replace(/[̀-ͯ]/g, "").toLowerCase(); }
  function filter() {
    var tokens = norm(search.value.trim()).split(/\s+/).filter(Boolean), visible = 0;
    sessions.forEach(function (s) {
      var kindOK = !type.value || s.dataset.kind === type.value;
      var cards = Array.prototype.slice.call(s.querySelectorAll(".corl-paper"));
      var any = false;
      if (cards.length) {
        var body = s.querySelector(".corl-body").cloneNode(true);
        Array.prototype.forEach.call(body.querySelectorAll(".corl-paper"), function (n) { n.remove(); });
        var base = norm(s.querySelector(".corl-time").textContent + " " + body.textContent);
        cards.forEach(function (c) {
          var hay = base + " " + norm(c.textContent);
          var m = tokens.every(function (t) { return hay.indexOf(t) !== -1; });
          c.hidden = !m; any = any || m;
        });
      } else {
        var hay = norm(s.textContent);
        any = tokens.every(function (t) { return hay.indexOf(t) !== -1; });
      }
      s.hidden = !(kindOK && any);
      if (!s.hidden) visible++;
    });
    Array.prototype.forEach.call(root.querySelectorAll(".corl-day"), function (d) {
      d.hidden = !Array.prototype.some.call(d.querySelectorAll(".corl-session"), function (s) { return !s.hidden; });
    });
    empty.hidden = visible > 0;
    status.textContent = (!search.value && !type.value) ? "Showing all sessions"
      : visible + " matching session" + (visible === 1 ? "" : "s");
  }
  function reset() { search.value = ""; type.value = ""; filter(); }
  search.addEventListener("input", filter);
  type.addEventListener("change", filter);
  document.getElementById("corl-reset").addEventListener("click", function () { reset(); search.focus(); });
  Array.prototype.forEach.call(root.querySelectorAll(".corl-nav a, .corl-see a"), function (a) { a.addEventListener("click", reset); });
  window.addEventListener("beforeprint", reset);
})();
</script>
